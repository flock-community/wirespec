package community.flock.wirespec.generated.model;

import community.flock.wirespec.java.Wirespec;

public record Tag (
  String value
) implements Wirespec.Refined {
  @Override
  public Boolean validate() {
    return java.util.regex.Pattern.compile("^[a-z]+$").matcher(value).find();
  }
  @Override
  public String toString() {
    return value;
  }
  @Override
  public String value() {
    return value;
  }
};

package community.flock.wirespec.generated.model;

import community.flock.wirespec.java.Wirespec;

public record Score (
  Long value
) implements Wirespec.Refined {
  @Override
  public Boolean validate() {
    return 0 <= value && value <= 100;
  }
  @Override
  public String toString() {
    return value.toString();
  }
  @Override
  public Long value() {
    return value;
  }
};

public record Label (
  Tag name,
  String code,
  java.util.List<String> aliases
) implements Wirespec.Shape {
  @Override
  public java.util.List<String> validate() {
    return java.util.stream.Stream.of((!name().validate() ? java.util.List.of("name") : java.util.List.<String>of()), (java.util.regex.Pattern.compile("^[A-Z]{3}$").matcher(code()).find() ? java.util.List.<String>of() : java.util.List.of("code")), java.util.stream.IntStream.range(0, aliases().size()).mapToObj(i -> (java.util.regex.Pattern.compile("^[a-z]+$").matcher(aliases().get(i)).find() ? java.util.List.<String>of() : java.util.List.of("aliases[" + i + "]"))).flatMap(java.util.Collection::stream).toList()).flatMap(java.util.Collection::stream).toList();
  }
};

public record Board (
  java.util.Optional<Tag> owner,
  java.util.Optional<java.util.List<Tag>> tags,
  java.util.Optional<java.util.List<java.util.Optional<Tag>>> slots,
  java.util.List<java.util.List<Score>> grid,
  java.util.Map<String, Tag> tagsByKey,
  java.util.Map<String, java.util.List<Score>> scoresByKey,
  java.util.Optional<Label> label,
  java.util.List<java.util.List<Label>> labels,
  java.util.Map<String, Label> labelsByKey,
  Long limit,
  java.util.List<Double> ratios
) implements Wirespec.Shape {
  @Override
  public java.util.List<String> validate() {
    return java.util.stream.Stream.of(owner().map(it -> (!it.validate() ? java.util.List.of("owner") : java.util.List.<String>of())).orElse(java.util.List.<String>of()), tags().map(it -> java.util.stream.IntStream.range(0, it.size()).mapToObj(i1 -> (!it.get(i1).validate() ? java.util.List.of("tags[" + i1 + "]") : java.util.List.<String>of())).flatMap(java.util.Collection::stream).toList()).orElse(java.util.List.<String>of()), slots().map(it -> java.util.stream.IntStream.range(0, it.size()).mapToObj(i1 -> it.get(i1).map(it2 -> (!it2.validate() ? java.util.List.of("slots[" + i1 + "]") : java.util.List.<String>of())).orElse(java.util.List.<String>of())).flatMap(java.util.Collection::stream).toList()).orElse(java.util.List.<String>of()), java.util.stream.IntStream.range(0, grid().size()).mapToObj(i -> java.util.stream.IntStream.range(0, grid().get(i).size()).mapToObj(i1 -> (!grid().get(i).get(i1).validate() ? java.util.List.of("grid[" + i + "][" + i1 + "]") : java.util.List.<String>of())).flatMap(java.util.Collection::stream).toList()).flatMap(java.util.Collection::stream).toList(), tagsByKey().keySet().stream().map(k -> (!tagsByKey().get(k).validate() ? java.util.List.of("tagsByKey[" + k + "]") : java.util.List.<String>of())).flatMap(java.util.Collection::stream).toList(), scoresByKey().keySet().stream().map(k -> java.util.stream.IntStream.range(0, scoresByKey().get(k).size()).mapToObj(i1 -> (!scoresByKey().get(k).get(i1).validate() ? java.util.List.of("scoresByKey[" + k + "][" + i1 + "]") : java.util.List.<String>of())).flatMap(java.util.Collection::stream).toList()).flatMap(java.util.Collection::stream).toList(), label().map(it -> it.validate().stream().map(e -> "label." + e).toList()).orElse(java.util.List.<String>of()), java.util.stream.IntStream.range(0, labels().size()).mapToObj(i -> java.util.stream.IntStream.range(0, labels().get(i).size()).mapToObj(i1 -> labels().get(i).get(i1).validate().stream().map(e -> "labels[" + i + "][" + i1 + "]." + e).toList()).flatMap(java.util.Collection::stream).toList()).flatMap(java.util.Collection::stream).toList(), labelsByKey().keySet().stream().map(k -> labelsByKey().get(k).validate().stream().map(e -> "labelsByKey[" + k + "]." + e).toList()).flatMap(java.util.Collection::stream).toList(), (1 <= limit() && limit() <= 10 ? java.util.List.<String>of() : java.util.List.of("limit")), java.util.stream.IntStream.range(0, ratios().size()).mapToObj(i -> (0.0 <= ratios().get(i) && ratios().get(i) <= 1.0 ? java.util.List.<String>of() : java.util.List.of("ratios[" + i + "]"))).flatMap(java.util.Collection::stream).toList()).flatMap(java.util.Collection::stream).toList();
  }
};

  }
};
