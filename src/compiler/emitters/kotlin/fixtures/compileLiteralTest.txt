package community.flock.wirespec.generated.model

import community.flock.wirespec.kotlin.Wirespec
import kotlin.reflect.typeOf

public data class Dog(
  val kind: String
) : Wirespec.Shape {
  override fun validate(): List<String> =
    if ((kind == "dog")) emptyList<String>() else listOf<String>("kind")
}

package community.flock.wirespec.generated.generator

import community.flock.wirespec.kotlin.Wirespec
import kotlin.reflect.typeOf
import community.flock.wirespec.generated.model.Dog

public object DogGenerator {
  public fun generate(generator: Wirespec.Generator, path: List<String>): Dog =
    Dog(kind = "dog")
}
