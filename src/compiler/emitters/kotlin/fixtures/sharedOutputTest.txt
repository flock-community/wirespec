package community.flock.wirespec.kotlin
import kotlin.reflect.KType
public object Wirespec {
  public interface Model
  public fun interface Shape : Model {
      public fun validate(): List<String>
  }
  public interface Enum : Model {
      public val label: String
  }
  public interface Refined<T: Any> : Model {
      public val value: T
      public fun validate(): Boolean
  }
  public interface Endpoint
  public interface Channel
  public interface Path
  public interface Queries
  public interface Headers
  public interface Handler
  public interface Call
  public enum class Method {
      GET,
      PUT,
      POST,
      DELETE,
      OPTIONS,
      HEAD,
      PATCH,
      TRACE
    }  public interface Request<T: Any> {
      public val path: Path
      public val method: Method
      public val queries: Queries
      public val headers: Headers
      public val body: T
      public interface Headers
  }
  public interface Response<T: Any> {
      public val status: Int
      public val headers: Headers
      public val body: T
      public interface Headers
  }
  public interface BodySerializer {
      public fun <T: Any> serializeBody(t: T, type: KType): ByteArray
  }
  public interface BodyDeserializer {
      public fun <T: Any> deserializeBody(raw: ByteArray, type: KType): T
  }
  public interface BodySerialization : BodySerializer, BodyDeserializer
  public interface PathSerializer {
      public fun <T: Any> serializePath(t: T, type: KType): String
  }
  public interface PathDeserializer {
      public fun <T: Any> deserializePath(raw: String, type: KType): T
  }
  public interface PathSerialization : PathSerializer, PathDeserializer
  public interface ParamSerializer {
      public fun <T: Any> serializeParam(value: T, type: KType): List<String>
  }
  public interface ParamDeserializer {
      public fun <T: Any> deserializeParam(values: List<String>, type: KType): T
  }
  public interface ParamSerialization : ParamSerializer, ParamDeserializer
  public interface Serializer : BodySerializer, PathSerializer, ParamSerializer
  public interface Deserializer : BodyDeserializer, PathDeserializer, ParamDeserializer
  public interface Serialization : Serializer, Deserializer
  public data class RawRequest(
      val method: String,
      val path: List<String>,
      val queries: Map<String, List<String>>,
      val headers: Map<String, List<String>>,
      val body: ByteArray?
    )
  public data class RawResponse(
      val statusCode: Int,
      val headers: Map<String, List<String>>,
      val body: ByteArray?
    )
  public fun interface Transportation {
      public suspend fun transport(request: RawRequest): RawResponse
  }
  public sealed interface GeneratorField<T: Any?>
  public data class GeneratorFieldString(
      val regex: String?,
      val annotations: List<Map<String, Any>>
    ) : GeneratorField<String>
  public data class GeneratorFieldInteger64(
      val min: Long?,
      val max: Long?,
      val annotations: List<Map<String, Any>>
    ) : GeneratorField<Long>
  public data class GeneratorFieldInteger32(
      val min: Int?,
      val max: Int?,
      val annotations: List<Map<String, Any>>
    ) : GeneratorField<Int>
  public data class GeneratorFieldNumber64(
      val min: Double?,
      val max: Double?,
      val annotations: List<Map<String, Any>>
    ) : GeneratorField<Double>
  public data class GeneratorFieldNumber32(
      val min: Float?,
      val max: Float?,
      val annotations: List<Map<String, Any>>
    ) : GeneratorField<Float>
  public data class GeneratorFieldBoolean(
      val annotations: List<Map<String, Any>>
    ) : GeneratorField<Boolean>
  public data class GeneratorFieldBytes(
      val annotations: List<Map<String, Any>>
    ) : GeneratorField<ByteArray>
  public data class GeneratorFieldEnum(
      val values: List<String>,
      val annotations: List<Map<String, Any>>,
      val type: KType
    ) : GeneratorField<String>
  public data class GeneratorFieldUnion(
      val variants: List<String>,
      val annotations: List<Map<String, Any>>,
      val type: KType
    ) : GeneratorField<String>
  public data class GeneratorFieldArray<T: Any>(
      val generate: (List<String>) -> T
    ) : GeneratorField<List<T>>
  public data class GeneratorFieldNullable<T: Any>(
      val generate: (List<String>) -> T
    ) : GeneratorField<T?>
  public data class GeneratorFieldShape<T: Any>(
      val annotations: Map<String, List<Map<String, Any>>>,
      val generate: (List<String>) -> T,
      val type: KType
    ) : GeneratorField<T>
  public data class GeneratorFieldDict<V: Any>(
      val generate: (List<String>) -> V
    ) : GeneratorField<Map<String, V>>
  public interface Generator {
      public fun <T: Any?> generate(path: List<String>, field: GeneratorField<T>): T
  }
  public interface ServerEdge<Req: Request<*>, Res: Response<*>> {
      public fun from(request: RawRequest): Req
      public fun to(response: Res): RawResponse
  }
  public interface ClientEdge<Req: Request<*>, Res: Response<*>> {
      public fun to(request: Req): RawRequest
      public fun from(response: RawResponse): Res
  }
  public interface Client<Req: Request<*>, Res: Response<*>> {
      public val pathTemplate: String
      public val method: String
      public fun client(serialization: Serialization): ClientEdge<Req, Res>
  }
  public interface Server<Req: Request<*>, Res: Response<*>> {
      public val pathTemplate: String
      public val method: String
      public fun server(serialization: Serialization): ServerEdge<Req, Res>
  }
}
