package community.flock.wirespec.generated.rpc

public object GetUser : Wirespec.Rpc {
  public sealed interface Response
  public data class Result(
      val value: User
    ) : Response
  public data class Error(
      val value: String
    ) : Response

      public suspend fun getUser(id: String): Response
  }
}

package community.flock.wirespec.generated.generator

import community.flock.wirespec.kotlin.Wirespec
import kotlin.reflect.typeOf
import community.flock.wirespec.generated.model.User

public object UserGenerator {
  public fun generate(generator: Wirespec.Generator, path: List<String>): User =
    User(name = generator.generate(path + listOf<String>("name"), Wirespec.GeneratorFieldString(
      regex = null,
      annotations = emptyList<Map<String, Any>>()
    )))
}
