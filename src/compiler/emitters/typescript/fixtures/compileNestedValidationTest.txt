export type Tag = string;

  /^[a-z]+$/g.test(value);

export type Score = number;
export const validateScore = (value: number) =>
  0 <= value && value <= 100;

import {Wirespec} from '../Wirespec'
import {Tag} from './Tag'
import {validateTag} from './Tag'

export type Label = {
  "name": Tag,
  "code": string,
  "aliases": string[],
}
export function validateLabel(obj: Label): string[] {
  return [...(!validateTag(obj.name) ? ['name'] : [] as string[]), ...(/^[A-Z]{3}$/g.test(obj.code) ? [] as string[] : ['code']), ...obj.aliases.flatMap((el, i) => (/^[a-z]+$/g.test(el) ? [] as string[] : [`aliases[${i}]`]))];
}

export type Board = {
  "owner": Tag | undefined,
  "tags": Tag[] | undefined,
  "slots": (Tag | undefined)[] | undefined,
  "grid": Score[][],
  "tagsByKey": Record<string, Tag>,
  "scoresByKey": Record<string, Score[]>,
  "label": Label | undefined,
  "labels": Label[][],
  "labelsByKey": Record<string, Label>,
  "limit": number,
  "ratios": number[],
}
export function validateBoard(obj: Board): string[] {
  return [...obj.owner != null ? (!validateTag(obj.owner) ? ['owner'] : [] as string[]) : [] as string[], ...obj.tags != null ? obj.tags.flatMap((el1, i1) => (!validateTag(el1) ? [`tags[${i1}]`] : [] as string[])) : [] as string[], ...obj.slots != null ? obj.slots.flatMap((el1, i1) => el1 != null ? (!validateTag(el1) ? [`slots[${i1}]`] : [] as string[]) : [] as string[]) : [] as string[], ...obj.grid.flatMap((el, i) => el.flatMap((el1, i1) => (!validateScore(el1) ? [`grid[${i}][${i1}]`] : [] as string[]))), ...Object.entries(obj.tagsByKey).flatMap(([k, v]) => (!validateTag(v) ? [`tagsByKey[${k}]`] : [] as string[])), ...Object.entries(obj.scoresByKey).flatMap(([k, v]) => v.flatMap((el1, i1) => (!validateScore(el1) ? [`scoresByKey[${k}][${i1}]`] : [] as string[]))), ...obj.label != null ? validateLabel(obj.label).map(e => `label.${e}`) : [] as string[], ...obj.labels.flatMap((el, i) => el.flatMap((el1, i1) => validateLabel(el1).map(e => `labels[${i}][${i1}].${e}`))), ...Object.entries(obj.labelsByKey).flatMap(([k, v]) => validateLabel(v).map(e => `labelsByKey[${k}].${e}`)), ...(1 <= obj.limit && obj.limit <= 10 ? [] as string[] : ['limit']), ...obj.ratios.flatMap((el, i) => (0.0 <= el && el <= 1.0 ? [] as string[] : [`ratios[${i}]`]))];
}

export type TreeNode = {
  "tag": Tag,
  "children": TreeNode[],
}
export function validateTreeNode(obj: TreeNode): string[] {
  return [...(!validateTag(obj.tag) ? ['tag'] : [] as string[]), ...obj.children.flatMap((el, i) => validateTreeNode(el).map(e => `children[${i}].${e}`))];
}

export namespace TagGenerator {
  export function generate(generator: Wirespec.Generator, path: string[]): Tag {
    return generator.generate([...path, ...['value']], { kind: 'string', regex: '^[a-z]+$', annotations: [] as Record<string, any>[] });
  }
}

}

export namespace LabelGenerator {
  export function generate(generator: Wirespec.Generator, path: string[]): Label {
    return { name: generator.generate([...path, ...['name']], { kind: 'shape', annotations: {}, generate: (p0: string[]) => TagGenerator.generate(generator, p0), type: "Tag" }), code: generator.generate([...path, ...['code']], { kind: 'string', regex: '^[A-Z]{3}$', annotations: [] as Record<string, any>[] }), aliases: generator.generate([...path, ...['aliases']], { kind: 'array', generate: (p0: string[]) => generator.generate(p0, { kind: 'string', regex: '^[a-z]+$', annotations: [] as Record<string, any>[] }) }) };
  }
}
