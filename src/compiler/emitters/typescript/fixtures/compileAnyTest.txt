import {Wirespec} from '../Wirespec'
export type Message = {
  "payload": any,
  "metadata": any | undefined,
  "attachments": any[],
  "extensions": Record<string, any>,
}
export function validateMessage(obj: Message): string[] {
  return [] as string[];
}

import {Wirespec} from '../Wirespec'
import {Message} from '../model/Message'
export namespace MessageGenerator {
  export function generate(generator: Wirespec.Generator, path: string[]): Message {
    return { payload: generator.generate([...path, ...['payload']], { kind: 'string', regex: undefined, annotations: [] as Record<string, any>[] }), metadata: generator.generate([...path, ...['metadata']], { kind: 'nullable', generate: (p0: string[]) => generator.generate(p0, { kind: 'string', regex: undefined, annotations: [] as Record<string, any>[] }) }), attachments: generator.generate([...path, ...['attachments']], { kind: 'array', generate: (p0: string[]) => generator.generate(p0, { kind: 'string', regex: undefined, annotations: [] as Record<string, any>[] }) }), extensions: generator.generate([...path, ...['extensions']], { kind: 'dict', generate: (p0: string[]) => generator.generate(p0, { kind: 'string', regex: undefined, annotations: [] as Record<string, any>[] }) }) };
  }
}

export {Message} from './Message'
