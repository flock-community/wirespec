export type Cat = {
  "kind": "cat",
  "name": string,
}
export function validateCat(obj: Cat): string[] {
  return ((obj.kind === 'cat') ? [] as string[] : ['kind']);
}

import {Wirespec} from '../Wirespec'

export type Dog = {
  "kind": "dog",
  "nickname": "good \"boy\"" | undefined,
  "tags": "pet"[],
  "tagsByKey": Record<string, "pet">,
}
export function validateDog(obj: Dog): string[] {
  return [...((obj.kind === 'dog') ? [] as string[] : ['kind']), ...obj.nickname != null ? ((obj.nickname === 'good "boy"') ? [] as string[] : ['nickname']) : [] as string[], ...obj.tags.flatMap((el, i) => ((el === 'pet') ? [] as string[] : [`tags[${i}]`])), ...Object.entries(obj.tagsByKey).flatMap(([k, v]) => ((v === 'pet') ? [] as string[] : [`tagsByKey[${k}]`]))];
}

import {Wirespec} from '../Wirespec'
import {type Dog} from '../model'

export namespace GetDogs {
  export type Path = {}
  export type Queries = {
    "sort": "name",
  }
  export type RequestHeaders = {}
  export type Request = {
    "path": Path,
    "method": Wirespec.Method,
    "queries": Queries,
    "headers": RequestHeaders,
    "body": void,
  }
  export type RequestParams = {"sort": "name"}
  export const request = (params: RequestParams): Request => ({
    path: {},
    method: "GET",
    queries: {"sort": params["sort"]},
    headers: {},
    body: undefined,
  })
  export type Response<T = unknown> = Response2XX | ResponseListDog
  export type Response2XX<T = unknown> = Response200
  export type ResponseListDog = Response200
  export type Response200Headers = {}
  export type Response200 = {
    "status": 200,
    "headers": Response200Headers,
    "body": Dog[],
  }
  export type Response200Params = {"body": Dog[]}
  export const response200 = (params: Response200Params): Response200 => ({
    status: 200,
    headers: {},
    body: params.body,
  })
  export interface Handler extends Wirespec.Handler {
    getDogs(_request: Request): Promise<Response<unknown>>;
  }
  export interface Call extends Wirespec.Call {
    getDogs(params: RequestParams): Promise<Response<unknown>>;
  }
  export function toRawRequest(serialization: Wirespec.Serializer, _request: Request): Wirespec.RawRequest {
    return { method: _request.method, path: ['dogs'], queries: { 'sort': serialization.serializeParam(_request.queries.sort, "string") }, headers: {}, body: undefined };
  }
  export function fromRawRequest(serialization: Wirespec.Deserializer, _request: Wirespec.RawRequest): Request {
    return request({"sort": _request.queries['sort'] != null ? serialization.deserializeParam(_request.queries['sort'], "string") : (() => { throw new Error('Param sort cannot be null') })()});
  }
  export function toRawResponse(serialization: Wirespec.Serializer, response: Response<unknown>): Wirespec.RawResponse {
    switch (response.status) {
      case 200: {
        const r = response as Response200;
        return { statusCode: r.status, headers: {}, body: serialization.serializeBody(r.body, "Dog[]") };
      }
      default: {
        throw new Error('Cannot match response with status:');
      }
    }
  }
  export function fromRawResponse(serialization: Wirespec.Deserializer, response: Wirespec.RawResponse): Response<unknown> {
    switch (response.statusCode) {
      case 200:
        return response200({"body": response.body != null ? serialization.deserializeBody(response.body, "Dog[]") : (() => { throw new Error('body is null') })()});
        break;
      default:
        throw new Error('Cannot match response with status:');
    }
  }
  export const client:Wirespec.Client<Request, Response> = (serialization: Wirespec.Serialization) => ({
    from: (it) => fromRawResponse(serialization, it),
    to: (it) => toRawRequest(serialization, it)
  })
  export const server:Wirespec.Server<Request, Response> = (serialization: Wirespec.Serialization) => ({
    from: (it) => fromRawRequest(serialization, it),
    to: (it) => toRawResponse(serialization, it)
  })
  export const api = {
    name: "getDogs",
    method: "GET",
    path: "dogs",
    server,
    client
  } as const
}

import {Wirespec} from '../Wirespec'
import {GetDogs} from '../endpoint/GetDogs'
import {type Dog} from '../model'

export const getDogsClient = (serialization: Wirespec.Serialization, transportation: Wirespec.Transportation) => ({
  getDogs: async (params: GetDogs.RequestParams): Promise<GetDogs.Response<unknown>> => {
    const request: GetDogs.Request = GetDogs.request(params);
    const rawRequest = GetDogs.toRawRequest(serialization, request);
    const rawResponse = await transportation.transport(rawRequest);
    return GetDogs.fromRawResponse(serialization, rawResponse);
  }
})

export namespace CatGenerator {
  export function generate(generator: Wirespec.Generator, path: string[]): Cat {
    return { kind: 'cat', name: generator.generate([...path, ...['name']], { kind: 'string', regex: undefined, annotations: [] as Record<string, any>[] }) };
  }
}

import {Wirespec} from '../Wirespec'
import {Dog} from '../model/Dog'

export namespace DogGenerator {
  export function generate(generator: Wirespec.Generator, path: string[]): Dog {
    return { kind: 'dog', nickname: generator.generate([...path, ...['nickname']], { kind: 'nullable', generate: (p0: string[]) => 'good "boy"' }), tags: generator.generate([...path, ...['tags']], { kind: 'array', generate: (p0: string[]) => 'pet' }), tagsByKey: generator.generate([...path, ...['tagsByKey']], { kind: 'dict', generate: (p0: string[]) => 'pet' }) };
  }
}
