    pub name: String,
}

        return Vec::<String>::new();
    }
}

use super::super::wirespec::*;
use regex;
use super::super::model::user::User;
pub struct UserGenerator;
impl UserGenerator {
    pub fn generate(generator: Wirespec.Generator, path: Vec<String>) -> User {
        return User { name: generator.generate(vec![path.as_slice(), vec![String::from("name")].as_slice()].concat(), Wirespec.GeneratorFieldString { regex: None, annotations: Vec::<std::collections::HashMap<String, Box<dyn std::any::Any>>>::new() }) };
    }
}
