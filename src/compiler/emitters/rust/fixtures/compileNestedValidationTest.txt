#[derive(Debug, Clone, Default, PartialEq)]
pub struct Tag {
    pub value: String,
}
impl Tag {
    pub fn validate(&self) -> bool {
        return regex::Regex::new(r"^[a-z]+$").unwrap().is_match(&self.value);
    }
    pub fn to_string(&self) -> String {
        return self.value.clone();
    }
}

use super::super::wirespec::*;
use regex;

#[derive(Debug, Clone, Default, PartialEq)]
pub struct Score {
    pub value: i64,
}
impl Score {
    pub fn validate(&self) -> bool {
        return 0 <= self.value && self.value <= 100;
    }
    pub fn to_string(&self) -> String {
        return format!("{}", self.value);
    }
}

#[derive(Debug, Clone, Default, PartialEq)]
pub struct Label {
    pub name: Tag,
    pub code: String,
    pub aliases: Vec<String>,
}
impl Label {
    pub fn validate(&self) -> Vec<String> {
        return vec![if !self.name.validate() { vec![String::from("name")] } else { Vec::<String>::new() }.as_slice(), if regex::Regex::new(r"^[A-Z]{3}$").unwrap().is_match(&self.code) { Vec::<String>::new() } else { vec![String::from("code")] }.as_slice(), self.aliases.iter().enumerate().flat_map(|(i, el)| if regex::Regex::new(r"^[a-z]+$").unwrap().is_match(&el) { Vec::<String>::new() } else { vec![format!("aliases[{}]", i)] }).collect::<Vec<_>>().as_slice()].concat();
    }
}

#[derive(Debug, Clone, Default, PartialEq)]
pub struct Board {
    pub owner: Option<Tag>,
    pub tags: Option<Vec<Tag>>,
    pub slots: Option<Vec<Option<Tag>>>,
    pub grid: Vec<Vec<Score>>,
    pub tags_by_key: std::collections::HashMap<String, Tag>,
    pub scores_by_key: std::collections::HashMap<String, Vec<Score>>,
    pub label: Option<Label>,
    pub labels: Vec<Vec<Label>>,
    pub labels_by_key: std::collections::HashMap<String, Label>,
    pub limit: i64,
    pub ratios: Vec<f64>,
}
impl Board {
    pub fn validate(&self) -> Vec<String> {
        return vec![self.owner.as_ref().map(|it| if !it.validate() { vec![String::from("owner")] } else { Vec::<String>::new() }).unwrap_or(Vec::<String>::new()).as_slice(), self.tags.as_ref().map(|it| it.iter().enumerate().flat_map(|(i1, el1)| if !el1.validate() { vec![format!("tags[{}]", i1)] } else { Vec::<String>::new() }).collect::<Vec<_>>()).unwrap_or(Vec::<String>::new()).as_slice(), self.slots.as_ref().map(|it| it.iter().enumerate().flat_map(|(i1, el1)| el1.as_ref().map(|it2| if !it2.validate() { vec![format!("slots[{}]", i1)] } else { Vec::<String>::new() }).unwrap_or(Vec::<String>::new())).collect::<Vec<_>>()).unwrap_or(Vec::<String>::new()).as_slice(), self.grid.iter().enumerate().flat_map(|(i, el)| el.iter().enumerate().flat_map(|(i1, el1)| if !el1.validate() { vec![format!("grid[{}][{}]", i, i1)] } else { Vec::<String>::new() }).collect::<Vec<_>>()).collect::<Vec<_>>().as_slice(), self.tags_by_key.iter().flat_map(|(k, v)| if !v.validate() { vec![format!("tagsByKey[{}]", k)] } else { Vec::<String>::new() }).collect::<Vec<_>>().as_slice(), self.scores_by_key.iter().flat_map(|(k, v)| v.iter().enumerate().flat_map(|(i1, el1)| if !el1.validate() { vec![format!("scoresByKey[{}][{}]", k, i1)] } else { Vec::<String>::new() }).collect::<Vec<_>>()).collect::<Vec<_>>().as_slice(), self.label.as_ref().map(|it| it.validate().iter().map(|e| format!("label.{}", e)).collect::<Vec<_>>()).unwrap_or(Vec::<String>::new()).as_slice(), self.labels.iter().enumerate().flat_map(|(i, el)| el.iter().enumerate().flat_map(|(i1, el1)| el1.validate().iter().map(|e| format!("labels[{}][{}].{}", i, i1, e)).collect::<Vec<_>>()).collect::<Vec<_>>()).collect::<Vec<_>>().as_slice(), self.labels_by_key.iter().flat_map(|(k, v)| v.validate().iter().map(|e| format!("labelsByKey[{}].{}", k, e)).collect::<Vec<_>>()).collect::<Vec<_>>().as_slice(), if 1 <= self.limit && self.limit <= 10 { Vec::<String>::new() } else { vec![String::from("limit")] }.as_slice(), self.ratios.iter().enumerate().flat_map(|(i, el)| if 0.0 <= el.clone() && el.clone() <= 1.0 { Vec::<String>::new() } else { vec![format!("ratios[{}]", i)] }).collect::<Vec<_>>().as_slice()].concat();
    }
}

#[derive(Debug, Clone, Default, PartialEq)]
pub struct TreeNode {
    pub tag: Tag,
    pub children: Vec<TreeNode>,
}
impl TreeNode {
    pub fn validate(&self) -> Vec<String> {
        return vec![if !self.tag.validate() { vec![String::from("tag")] } else { Vec::<String>::new() }.as_slice(), self.children.iter().enumerate().flat_map(|(i, el)| el.validate().iter().map(|e| format!("children[{}].{}", i, e)).collect::<Vec<_>>()).collect::<Vec<_>>().as_slice()].concat();
    }
}
