use std::any::TypeId;
use std::collections::HashMap;
pub trait Model {}
pub trait Shape: Model {
    fn validate(&self) -> Vec<String>;
}
pub trait Enum: Sized {
    fn label(&self) -> &str;
    fn from_label(s: &str) -> Option<Self>;
}
pub trait Refined<T> {
    fn value(&self) -> &T;
    fn validate(&self) -> bool;
}
pub trait Endpoint {}
pub trait Channel {}
pub trait Rpc {}
pub trait Path {}
pub trait Queries {}
pub trait Headers {}
pub trait Handler {}
pub trait Call {}
#[derive(Debug, Clone, Default, PartialEq)]
pub enum Method {
    #[default]
    GET,
    PUT,
    POST,
    DELETE,
    OPTIONS,
    HEAD,
    PATCH,
    TRACE,
}
pub trait Request<T> {
    fn path(&self) -> &dyn Path;
    fn method(&self) -> &Method;
    fn queries(&self) -> &dyn Queries;
    fn headers(&self) -> &dyn RequestHeaders;
    fn body(&self) -> &T;
}
pub trait RequestHeaders: Headers {}
pub trait Response<T> {
    fn status(&self) -> i32;
    fn headers(&self) -> &dyn ResponseHeaders;
    fn body(&self) -> &T;
}
pub trait ResponseHeaders: Headers {}
pub trait BodySerializer {
    fn serialize_body<T: 'static>(&self, t: &T, r#type: TypeId) -> Vec<u8>;
}
pub trait BodyDeserializer {
    fn deserialize_body<T: 'static>(&self, raw: &[u8], r#type: TypeId) -> T;
}
pub trait BodySerialization: BodySerializer + BodyDeserializer {}
pub trait PathSerializer {
    fn serialize_path<T: std::fmt::Display>(&self, t: &T, r#type: TypeId) -> String;
}
pub trait PathDeserializer {
    fn deserialize_path<T: std::str::FromStr>(&self, raw: &str, r#type: TypeId) -> T where T::Err: std::fmt::Debug;
}
pub trait PathSerialization: PathSerializer + PathDeserializer {}
pub trait ParamSerializer {
    fn serialize_param<T: 'static>(&self, value: &T, r#type: TypeId) -> Vec<String>;
}
pub trait ParamDeserializer {
    fn deserialize_param<T: 'static>(&self, values: &[String], r#type: TypeId) -> T;
}
pub trait ParamSerialization: ParamSerializer + ParamDeserializer {}
pub trait Serializer: BodySerializer + PathSerializer + ParamSerializer {}
pub trait Deserializer: BodyDeserializer + PathDeserializer + ParamDeserializer {}
pub trait Serialization: Serializer + Deserializer {}
#[derive(Debug, Clone, PartialEq)]
pub struct RawRequest {
    pub method: String,
    pub path: Vec<String>,
    pub queries: std::collections::HashMap<String, Vec<String>>,
    pub headers: std::collections::HashMap<String, Vec<String>>,
    pub body: Option<Vec<u8>>,
}
#[derive(Debug, Clone, PartialEq)]
pub struct RawResponse {
    pub status_code: i32,
    pub headers: std::collections::HashMap<String, Vec<String>>,
    pub body: Option<Vec<u8>>,
}
pub trait Transportation {
    async fn transport(&self, request: &RawRequest) -> RawResponse;
}
pub type Annotations = Vec<HashMap<String, Box<dyn std::any::Any>>>;
pub trait GeneratorField<T> {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> T;
}
pub struct GeneratorFieldString {
    pub regex: Option<String>,
    pub annotations: Annotations,
}
pub struct GeneratorFieldInteger64 {
    pub min: Option<i64>,
    pub max: Option<i64>,
    pub annotations: Annotations,
}
pub struct GeneratorFieldInteger32 {
    pub min: Option<i32>,
    pub max: Option<i32>,
    pub annotations: Annotations,
}
pub struct GeneratorFieldNumber64 {
    pub min: Option<f64>,
    pub max: Option<f64>,
    pub annotations: Annotations,
}
pub struct GeneratorFieldNumber32 {
    pub min: Option<f32>,
    pub max: Option<f32>,
    pub annotations: Annotations,
}
pub struct GeneratorFieldBoolean {
    pub annotations: Annotations,
}
pub struct GeneratorFieldBytes {
    pub annotations: Annotations,
}
pub struct GeneratorFieldEnum {
    pub values: Vec<String>,
    pub annotations: Annotations,
    pub r#type: TypeId,
}
pub struct GeneratorFieldUnion {
    pub variants: Vec<String>,
    pub annotations: Annotations,
    pub r#type: TypeId,
}
pub struct GeneratorFieldArray<'a, T> {
    pub generate: Box<dyn Fn(Vec<String>) -> T + 'a>,
}
pub struct GeneratorFieldNullable<'a, T> {
    pub generate: Box<dyn Fn(Vec<String>) -> T + 'a>,
}
pub struct GeneratorFieldDict<'a, T> {
    pub generate: Box<dyn Fn(Vec<String>) -> T + 'a>,
}
pub struct GeneratorFieldShape<'a, T> {
    pub annotations: HashMap<String, Annotations>,
    pub generate: Box<dyn Fn(Vec<String>) -> T + 'a>,
    pub r#type: TypeId,
}
impl GeneratorField<String> for GeneratorFieldString {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> String {
        generator.generate_string(path, self)
    }
}
impl GeneratorField<i64> for GeneratorFieldInteger64 {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> i64 {
        generator.generate_integer64(path, self)
    }
}
impl GeneratorField<i32> for GeneratorFieldInteger32 {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> i32 {
        generator.generate_integer32(path, self)
    }
}
impl GeneratorField<f64> for GeneratorFieldNumber64 {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> f64 {
        generator.generate_number64(path, self)
    }
}
impl GeneratorField<f32> for GeneratorFieldNumber32 {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> f32 {
        generator.generate_number32(path, self)
    }
}
impl GeneratorField<bool> for GeneratorFieldBoolean {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> bool {
        generator.generate_boolean(path, self)
    }
}
impl GeneratorField<Vec<u8>> for GeneratorFieldBytes {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> Vec<u8> {
        generator.generate_bytes(path, self)
    }
}
impl GeneratorField<String> for GeneratorFieldEnum {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> String {
        generator.generate_enum(path, self)
    }
}
impl GeneratorField<String> for GeneratorFieldUnion {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> String {
        generator.generate_union(path, self)
    }
}
impl<'a, T> GeneratorField<Vec<T>> for GeneratorFieldArray<'a, T> {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> Vec<T> {
        generator.generate_array(path, self)
    }
}
impl<'a, T> GeneratorField<Option<T>> for GeneratorFieldNullable<'a, T> {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> Option<T> {
        generator.generate_nullable(path, self)
    }
}
impl<'a, T> GeneratorField<HashMap<String, T>> for GeneratorFieldDict<'a, T> {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> HashMap<String, T> {
        generator.generate_dict(path, self)
    }
}
impl<'a, T> GeneratorField<T> for GeneratorFieldShape<'a, T> {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> T {
        generator.generate_shape(path, self)
    }
}
// A field of type `Any` is generated as a string.
impl GeneratorField<AnyValue> for GeneratorFieldString {
    fn generate(&self, generator: &impl Generator, path: Vec<String>) -> AnyValue {
        AnyValue::new(generator.generate_string(path, self))
    }
}
pub trait Generator: Sized {
    fn generate_string(&self, path: Vec<String>, field: &GeneratorFieldString) -> String;
    fn generate_integer64(&self, path: Vec<String>, field: &GeneratorFieldInteger64) -> i64;
    fn generate_integer32(&self, path: Vec<String>, field: &GeneratorFieldInteger32) -> i32;
    fn generate_number64(&self, path: Vec<String>, field: &GeneratorFieldNumber64) -> f64;
    fn generate_number32(&self, path: Vec<String>, field: &GeneratorFieldNumber32) -> f32;
    fn generate_boolean(&self, path: Vec<String>, field: &GeneratorFieldBoolean) -> bool;
    fn generate_bytes(&self, path: Vec<String>, field: &GeneratorFieldBytes) -> Vec<u8>;
    fn generate_enum(&self, path: Vec<String>, field: &GeneratorFieldEnum) -> String;
    fn generate_union(&self, path: Vec<String>, field: &GeneratorFieldUnion) -> String;
    fn generate_array<T>(&self, path: Vec<String>, field: &GeneratorFieldArray<T>) -> Vec<T>;
    fn generate_nullable<T>(&self, path: Vec<String>, field: &GeneratorFieldNullable<T>) -> Option<T>;
    fn generate_dict<T>(&self, path: Vec<String>, field: &GeneratorFieldDict<T>) -> HashMap<String, T>;
    fn generate_shape<T>(&self, path: Vec<String>, field: &GeneratorFieldShape<T>) -> T;
    fn generate<T>(&self, path: Vec<String>, field: impl GeneratorField<T>) -> T {
        field.generate(self, path)
    }
}
#[derive(Debug, Clone)]
pub struct AnyValue(pub std::sync::Arc<dyn std::any::Any + Send + Sync>);
impl AnyValue {
    pub fn new<T: std::any::Any + Send + Sync>(value: T) -> Self {
        AnyValue(std::sync::Arc::new(value))
    }
    pub fn downcast_ref<T: std::any::Any>(&self) -> Option<&T> {
        self.0.downcast_ref::<T>()
    }
}
impl Default for AnyValue {
    fn default() -> Self {
        AnyValue::new(())
    }
}
impl PartialEq for AnyValue {
    fn eq(&self, other: &Self) -> bool {
        std::sync::Arc::ptr_eq(&self.0, &other.0)
    }
}
