@dataclass

@dataclass
class Label(Wirespec.Shape):
    name: Tag
    code: str
    aliases: list[str]
    def validate(self) -> list[str]:
        return (['name'] if not self.name.validate() else []) + ([] if bool(re.match(r"^[A-Z]{3}$", self.code)) else ['code']) + [item for i, el in enumerate(self.aliases) for item in ([] if bool(re.match(r"^[a-z]+$", el)) else [f"aliases[{i}]"])]

@dataclass
class Board(Wirespec.Shape):
    owner: Optional[Tag]
    tags: Optional[list[Tag]]
    slots: Optional[list[Optional[Tag]]]
    grid: list[list[Score]]
    tagsByKey: dict[str, Tag]
    scoresByKey: dict[str, list[Score]]
    label: Optional[Label]
    labels: list[list[Label]]
    labelsByKey: dict[str, Label]
    limit: int
    ratios: list[float]
    def validate(self) -> list[str]:
        return ((['owner'] if not self.owner.validate() else []) if self.owner is not None else []) + ([item for i1, el1 in enumerate(self.tags) for item in ([f"tags[{i1}]"] if not el1.validate() else [])] if self.tags is not None else []) + ([item for i1, el1 in enumerate(self.slots) for item in (([f"slots[{i1}]"] if not el1.validate() else []) if el1 is not None else [])] if self.slots is not None else []) + [item for i, el in enumerate(self.grid) for item in [item for i1, el1 in enumerate(el) for item in ([f"grid[{i}][{i1}]"] if not el1.validate() else [])]] + [item for k, v in self.tagsByKey.items() for item in ([f"tagsByKey[{k}]"] if not v.validate() else [])] + [item for k, v in self.scoresByKey.items() for item in [item for i1, el1 in enumerate(v) for item in ([f"scoresByKey[{k}][{i1}]"] if not el1.validate() else [])]] + ([f"label.{e}" for e in self.label.validate()] if self.label is not None else []) + [item for i, el in enumerate(self.labels) for item in [item for i1, el1 in enumerate(el) for item in [f"labels[{i}][{i1}].{e}" for e in el1.validate()]]] + [item for k, v in self.labelsByKey.items() for item in [f"labelsByKey[{k}].{e}" for e in v.validate()]] + ([] if 1 <= self.limit and self.limit <= 10 else ['limit']) + [item for i, el in enumerate(self.ratios) for item in ([] if 0.0 <= el and el <= 1.0 else [f"ratios[{i}]"])]
