@dataclass
class Path(Wirespec.Path):
    pass
@dataclass
class Queries(Wirespec.Queries):
    sort: str
@dataclass
class RequestHeaders(Wirespec.Request.Headers):
    pass
@dataclass
class Request(Wirespec.Request[None]):
    path: Path
    method: Wirespec.Method
    queries: Queries
    headers: RequestHeaders
    body: None
    def __init__(
        self,
        sort: str,
    ):
        self.path = Path()
        self.method = Wirespec.Method.GET
        self.queries = Queries(sort=sort)
        self.headers = RequestHeaders()
        self.body = None
class Response(Wirespec.Response[T], Generic[T]):
    pass
class Response2XX(Response[T], Generic[T]):
    pass
class ResponseListDog(Response[list[Dog]]):
    pass
@dataclass
class Response200Headers(Wirespec.Response.Headers):
    pass
@dataclass
class Response200(Response2XX[list[Dog]], ResponseListDog):
    status: int
    headers: Response200Headers
    body: list[Dog]
    def __init__(
        self,
        body: list[Dog],
    ):
        self.status = 200
        self.headers = Response200Headers()
        self.body = body
class GetDogs(Wirespec.Endpoint):
    class Handler(Wirespec.Handler, ABC):
        @abstractmethod
        async def get_dogs(self, request: Request) -> Response[Any]:
            ...
    class Call(Wirespec.Call, ABC):
        @abstractmethod
        async def get_dogs(self, sort: str) -> Response[Any]:
            ...
    @staticmethod
    def toRawRequest(serialization: Wirespec.Serializer, request: Request) -> Wirespec.RawRequest:
        return Wirespec.RawRequest(method=request.method.value, path=['dogs'], queries={'sort': serialization.serializeParam(request.queries.sort, str)}, headers={}, body=None)
    @staticmethod
    def fromRawRequest(serialization: Wirespec.Deserializer, request: Wirespec.RawRequest) -> Request:
        return Request(sort=(serialization.deserializeParam(request.queries['sort'], str) if request.queries['sort'] is not None else _raise('Param sort cannot be null')))
    @staticmethod
    def toRawResponse(serialization: Wirespec.Serializer, response: Response[Any]) -> Wirespec.RawResponse:
        match response:
            case Response200() as r:
                return Wirespec.RawResponse(statusCode=r.status, headers={}, body=serialization.serializeBody(r.body, list[Dog]))
            case _:
                raise Exception(('Cannot match response with status: ' + str(response.status)))
    @staticmethod
    def fromRawResponse(serialization: Wirespec.Deserializer, response: Wirespec.RawResponse) -> Response[Any]:
        match response.statusCode:
            case 200:
                return Response200(body=(serialization.deserializeBody(response.body, list[Dog]) if response.body is not None else _raise('body is null')))
            case _:
                raise Exception(('Cannot match response with status: ' + str(response.statusCode)))

@dataclass
class GetDogsClient(GetDogs.Call):
    serialization: Wirespec.Serialization
    transportation: Wirespec.Transportation
    async def get_dogs(self, sort: str) -> Response[Any]:
        request = Request(sort=sort)
        rawRequest = GetDogs.toRawRequest(serialization=self.serialization, request=request)
        rawResponse = await self.transportation.transport(rawRequest)
